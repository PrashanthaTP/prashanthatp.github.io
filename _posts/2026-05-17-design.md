```cpp
#include <iostream>
#include <initializer_list>

using std::cout;

template <typename T>
class VectorTest {
private:
    T *_data = nullptr;
    size_t _cap  = 0;
    size_t _size = 0;
    
    void grow(int newSize){
        if(newSize <= _cap){
            return;
        }
        
        T *newData = new T[newSize];
        for(size_t i=0; i<_size; i++){
            newData[i] = std::move(_data[i]);
        }
        delete[] _data;
        _data = newData;
        _cap = newSize;
    }
    
public:

    VectorTest(size_t n): _data(new T[n]),_size(n),_cap(n) {
        
    }
    
    VectorTest():_data(nullptr), _size(0), _cap(0){
        
    }
    
    VectorTest(std::initializer_list<T> list){
        size_t reqSize = list.size();
        grow(reqSize);
        _size = 0;
        for(auto item: list){
            _data[_size++] = item;
        }
    }
    
    ~VectorTest(){
        delete[] _data;
    }
    
    VectorTest(const VectorTest& other) = delete;
    VectorTest& operator=(const VectorTest& other) = delete;
        
    VectorTest(VectorTest&& other) = delete;
    VectorTest& operator=(VectorTest&& other) = delete;
    
    void push_back(T item) {
        if(_size == _cap){
            grow(_cap == 0? 1: _cap*2);
        }
        _data[_size++] = item;
    }
    
    void pop_back(){
        if(_size == 0){
            return;
        }
        _size-=1;
    }
    
    T& operator[](size_t index){
        return _data[index];
    }
    
    const T& operator[](size_t index) const{
        return _data[index];
    }
    
    size_t size() const{
        return _size;
    }
    
    T *begin(){
        return _data;
    }
        
    const T *begin() const{
        return _data;
    }
    
    T *end(){
        return _data + _size;
    }
    
    const T *end() const{
        return _data + _size;
    }
    
};

#define PRINT_STMT(x) cout << #x << "\n"; x
#define PRINT_VEC(v) for(auto x: v ) {cout << x << " | ";} cout << "\n"

int main(){
    VectorTest<int> vec1{1,2,3};
    VectorTest<int> vec2(2);
    vec2[0] = 10;
    vec2[1] = 100;
    vec2.push_back(1000);
    vec2.push_back(10000);
```

```cpp
#include <iostream>

template <typename T>
class UniquePtr {
private:
    T *_ptr;

public:
    explicit UniquePtr(T *oPtr = nullptr) : _ptr(oPtr) {}
    ~UniquePtr() {
        delete _ptr;
    }
    
    // delete copy constructor and copy assignment operators
    UniquePtr(const UniquePtr& other) = delete;
    UniquePtr& operator=(const UniquePtr& other) = delete;
    
    UniquePtr(UniquePtr&& other): _ptr(other._ptr){
        other._ptr = nullptr;
    }
    UniquePtr& operator=(UniquePtr&& other){
        if(this != &other){
            delete _ptr;
            _ptr = other._ptr;
            other._ptr = nullptr;
        }
        return *this;
    }
    
    T& operator*() const{
        return *_ptr;
    }
    
    T* operator->() const{
        return _ptr;
    }
    
    T* get() const{
        return _ptr;
    }
    
    T* release(){
        T *temp = _ptr;
        _ptr = nullptr;
        return temp;
    }
    
    void reset(T *other = nullptr){
        if(_ptr != other){
            delete _ptr;
            _ptr = other;
        }
    }
    
};


struct Resource {
    void print(){
        std::cout << "Resource print\n";
    }
};


int main(){
    UniquePtr<Resource> ptr(new Resource);
    ptr->print();
    
    UniquePtr<Resource> ptr2 = std::move(ptr);
    if(ptr.get() == nullptr){
        std::cout << "Ptr got moved properly\n";
    }
    
    ptr2->print();
    
}
```
