```cpp
class Solution {
public:
    int getSum(int a, int b) {
        //https://leetcode.com/problems/sum-of-two-integers/solutions/132479/simple-explanation-on-how-to-arrive-at-t-p0qs
        int carry = 0;
        int sum = 0;
        while(/*1*/ b!=0/*b is carray after one iteration*/) {
            sum = a ^ b;
            // for python (a&b<<1) & OXFFFFFFFF; // to make it 32 bit
            // as python has unlimited precision, carry might not become zero if dont truncate it to 32 bits
            carry = (a & b)<<1; 
            //new a
            a = sum;
            //new b
            b = carry;
        }
        return a;// a is sum if b isn't zero
    }
};
```

```cpp
class Solution {
public:
    int hammingWeight(int n) {
        //Brian Kernighan’s algorithm
         int bits = 0;
         while(n){
            n = n & n-1;
            bits++;
         }
         return bits;
    }
};
```
